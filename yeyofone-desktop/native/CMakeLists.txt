cmake_minimum_required(VERSION 4.1.2)
project(yeyofone_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED CONFIG
  PATHS "${OPENSSL_ROOT_DIR}/lib/cmake/OpenSSL" NO_DEFAULT_PATH)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
set(PJLIB_WITH_SSL openssl CACHE STRING "" FORCE)
set(PJSIP_WITH_TLS ON CACHE BOOL "" FORCE)
set(PJNATH_WITH_UPNP OFF CACHE BOOL "" FORCE)
set(PJLIB_WITH_LIBUUID OFF CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_VIDEO OFF CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_SRTP ON CACHE BOOL "" FORCE)
set(SRTP_WITH_OPENSSL ON CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_RESAMPLE libresample CACHE STRING "" FORCE)
set(PJMEDIA_WITH_G711_CODEC ON CACHE BOOL "" FORCE)
set(PJMEDIA_WITH_G722_CODEC ON CACHE BOOL "" FORCE)
foreach(feature L16 GSM SPEEX ILBC G7221 OPENCORE_AMRNB OPENCORE_AMRWB SILK OPUS BCG729 LYRA ANDROID_MEDIACODEC)
  set(PJMEDIA_WITH_${feature}_CODEC OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature SPEEX_AEC WEBRTC_AEC WEBRTC_AEC3)
  set(PJMEDIA_WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(PJMEDIA_WITH_AUDIODEV ON CACHE BOOL "" FORCE)
add_compile_definitions(PJSUA_MAX_ACC=32)
add_subdirectory("${PJPROJECT_SOURCE_DIR}" pjproject EXCLUDE_FROM_ALL)
if(NOT PJLIB_WITH_SSL STREQUAL "openssl" OR NOT PJSIP_WITH_TLS OR NOT PJMEDIA_WITH_SRTP)
  message(FATAL_ERROR "Required TLS/SRTP capabilities are missing")
endif()
if(APPLE AND NOT PJMEDIA_WITH_AUDIODEV_COREAUDIO)
  message(FATAL_ERROR "Required CoreAudio backend is missing")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT PJMEDIA_WITH_AUDIODEV_ALSA)
  message(FATAL_ERROR "Install ALSA development headers for this audio profile")
endif()
add_executable(yeyofone-native-smoke smoke.cpp)
target_link_libraries(yeyofone-native-smoke PRIVATE pjsua2 OpenSSL::SSL)
# PJSIP 2.17 CMake does not propagate all public/generated include paths.
target_include_directories(yeyofone-native-smoke PRIVATE
  "${PJPROJECT_SOURCE_DIR}/pjmedia/include"
  "${PJPROJECT_SOURCE_DIR}/pjnath/include"
  "${PJPROJECT_SOURCE_DIR}/pjlib-util/include"
  "${CMAKE_CURRENT_BINARY_DIR}/pjproject/pjsip/include"
  "${CMAKE_CURRENT_BINARY_DIR}/pjproject/pjmedia/include"
)
enable_testing()
add_test(NAME native-link-smoke COMMAND yeyofone-native-smoke)
set_tests_properties(native-link-smoke PROPERTIES TIMEOUT 15)

add_library(yeyofone-voip STATIC bridge.cpp)
target_link_libraries(yeyofone-voip PRIVATE pjsua2 OpenSSL::SSL)
get_target_property(YEYO_INCLUDES yeyofone-native-smoke INCLUDE_DIRECTORIES)
target_include_directories(yeyofone-voip PRIVATE ${YEYO_INCLUDES})
add_dependencies(yeyofone-native-smoke yeyofone-voip)

add_executable(yeyofone-bridge-tests bridge-tests.cpp)
target_link_libraries(yeyofone-bridge-tests PRIVATE pjsua2 OpenSSL::SSL)
target_include_directories(yeyofone-bridge-tests PRIVATE ${YEYO_INCLUDES})
add_dependencies(yeyofone-native-smoke yeyofone-bridge-tests)
add_test(NAME native-bridge-contract COMMAND yeyofone-bridge-tests)
set_tests_properties(native-bridge-contract PROPERTIES TIMEOUT 30)
